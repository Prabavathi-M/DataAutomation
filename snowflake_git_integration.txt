USE DBT_SAMPLES;

--create secret for git repo
CREATE OR REPLACE SECRET MY_GIT_DATA_ENGINEERING_REPO_SECRET
TYPE = PASSWORD
USERNAME = '<user_name>'
PASSWORD = "<github_pat>"

--create an api integration
CREATE API INTEGRATION MYDBT_API_INTEGRATION
API_PROVIDER = git_https_api
ENABLED = TRUE
API_ALLOWED_PREFIXES = ('<target_repo>')
--API_USER_AUTHENTICATION = (type = snoflake_github_app) --OAuth
COMMENT = 'INTERGRATION FOR MY DBT REPO';

ALTER API INTEGRATION MYDBT_API_INTEGRATION
SET ALLOWED_AUTHENTICATION_SECRETS = (MY_GIT_DATA_ENGINEERING_REPO_SECRET);

ALTER API INTEGRATION MYDBT_API_INTEGRATION
SET API_ALLOWED_PREFIXES = ('<target_repo>');

--create git repo in snowflake
CREATE GIT REPOSITORY MY_GIT_REPO
API_INTEGRATION = MYDBT_API_INTEGRATION
ORIGIN = '<target_repo>';

ALTER GIT REPOSITORY MY_GIT_REPO FETCH;

SHOW GIT REPOSITORIES;

SHOW BRANCHES IN GIT REPOSITORY MY_GIT_REPO;
